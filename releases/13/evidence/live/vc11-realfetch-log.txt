class Observer:
    def __init__(self, label):
        self.label = label
        self.hops = []
    @contextlib.contextmanager
    def stream(self, method, url, **kwargs):
        host = urlsplit(str(url)).hostname
        assert urlsplit(str(url)).scheme == 'https'
        assert host == 'huggingface.co' or host.endswith(('.huggingface.co', '.xethub.hf.co', '.hf.co')), host
        headers = httpx.Headers(kwargs.get('headers', {}))
        row = dict(hop_index=len(self.hops)+1, url=str(url), host=host,
                   follow_redirects=kwargs.get('follow_redirects'),
                   authorization_present='authorization' in headers, range=headers.get('range'))
        self.hops.append(row)
        print(self.label, 'STREAM', json.dumps(row), flush=True)
        assert not row['authorization_present']
        with httpx.stream(method, url, **kwargs) as response:
            row.update(status=response.status_code, content_length=response.headers.get('content-length'),
                       content_range=response.headers.get('content-range'), location=response.headers.get('location'))
            print(self.label, 'RESPONSE', json.dumps(row), flush=True)
